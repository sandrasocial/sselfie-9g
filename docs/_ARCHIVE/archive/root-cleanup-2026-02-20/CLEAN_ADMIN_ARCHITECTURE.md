```
└─ Scalable: Add new agents without code changes
```

## 🔌 HOW TO CONNECT GUMLOOP AGENTS TO YOUR ADMIN

### Option 1: Gumloop API (Recommended)
Gumloop has an API that lets you trigger agents and get results.

**Example: Call Agent 6 (Lead Qualification)**
```typescript
// In your Next.js API route: /app/api/admin/get-hot-leads/route.ts

export async function GET() {
  const response = await fetch('https://api.gumloop.com/v1/agents/your-agent-6-id/run', {
    method: 'POST',
    headers: {
      'Authorization': `Bearer ${process.env.GUMLOOP_API_KEY}`,
      'Content-Type': 'application/json'
    },
    body: JSON.stringify({
      // Any inputs the agent needs
    })
  })

  const leads = await response.json()
  return Response.json({ leads })
}
```

**Then in your UI:**
```typescript
// /app/admin/leads/page.tsx
const { leads } = await fetch('/api/admin/get-hot-leads').then(r => r.json())

return (
  <div>
    <h1>Hot Leads</h1>
    {leads.map(lead => (
      <LeadCard
        name={lead.name}
        score={lead.score}
        dm={lead.personalized_dm}
      />
    ))}
  </div>
)
```

**Advantages:**
- ✅ Simple API calls
- ✅ No complex logic in your code
- ✅ Cheap (Gumloop pricing, not direct LLM costs)
- ✅ Easy to update agents (just change in Gumloop)

## 📝 CODE EXAMPLES

### Example 1: Simple Agent Chat Component
```typescript
// /components/admin/agent-chat.tsx
'use client'

import { useState } from 'react'

export function AgentChat({ agentId, agentName }) {
  const [messages, setMessages] = useState([])
  const [input, setInput] = useState('')

  async function sendMessage() {
    // Add user message
    setMessages([...messages, { role: 'user', content: input }])

    // Call Gumloop agent via your API
    const response = await fetch('/api/admin/chat-with-agent', {
      method: 'POST',
      body: JSON.stringify({ agentId, message: input })
    })

    const data = await response.json()

    // Add agent response
    setMessages([...messages, { role: 'agent', content: data.response }])
    setInput('')
  }

  return (
    <div>
      <h2>{agentName}</h2>
      <div className="messages">
        {messages.map((msg, i) => (
          <div key={i} className={msg.role}>
            {msg.content}
          </div>
        ))}
      </div>
      <input
        value={input}
        onChange={(e) => setInput(e.target.value)}
        placeholder="Ask the agent..."
      />
      <button onClick={sendMessage}>Send</button>
    </div>
  )
}
```

### Example 2: API Route to Call Gumloop
```typescript
// /app/api/admin/chat-with-agent/route.ts

export async function POST(req: Request) {
  const { agentId, message } = await req.json()

  // Call Gumloop API
  const response = await fetch(`https://api.gumloop.com/v1/agents/${agentId}/chat`, {
    method: 'POST',
    headers: {
      'Authorization': `Bearer ${process.env.GUMLOOP_API_KEY}`,
      'Content-Type': 'application/json'
    },
    body: JSON.stringify({
      message,
      session_id: 'user-session-123' // Optional: maintain conversation context
    })
  })

  const data = await response.json()

  return Response.json({
    response: data.message,
    agentId,
    timestamp: new Date().toISOString()
  })
}
```
